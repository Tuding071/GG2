              private fun natural(a: String, b: String): Int {
                  val rx = Regex("\\d+|\\D+")
                  val x = rx.findAll(a.lowercase()).map { it.value }.toList()
                  val y = rx.findAll(b.lowercase()).map { it.value }.toList()
                  for (i in 0 until minOf(x.size, y.size)) {
                      val p = x[i]
                      val q = y[i]
                      val c = if (p[0].isDigit() && q[0].isDigit()) {
                          val pp = p.trimStart('0')
                          val qq = q.trimStart('0')
                          if (pp.length != qq.length) pp.length - qq.length else pp.compareTo(qq)

                  }

              fun requestRemove(r: HideRule) {
                  val k = keyOf(r)
                  if (closing.contains(k)) return
                  closing.add(k)
                  val job = Runnable {
                      jobs.remove(k)
                      closing.remove(k)
                      remove(r.host, r.sel, r.path)
                      toast(Gecko.appCtx, "Rule removed. Reload the page.")
                  }
                  jobs[k] = job
                  main.postDelayed(job, 2000L)
              }

                  val k = keyOf(r)
                  val job = jobs.remove(k)
                  if (job != null) main.removeCallbacks(job)
                  closing.remove(k)
              }

                      val r = parse(host, arr.opt(i)) ?: continue

                  while (keys.hasNext()) {
                      val h = keys.next()

                  val arr = all.optJSONArray(k) ?: JSONArray()

                      val r = parse(k, arr.opt(i)) ?: continue
                      if (r.sel == sel && r.path == p) return

              }

                  val all = readAll()
                  all.remove(hostKey(host))
                  sp().edit().putString("hide", all.toString()).apply()

              }

                  ".b{max-width:420px;padding:28px}h1{font-size:22px;margin:0 0 12px}" +
                  "p{color:#AAAAAA;font-size:15px;line-height:1.45;margin:0 0 10px}" +
                  ".u{word-break:break-all;font-size:13px;color:#888888}" +
                  "button{margin-top:18px;background:#4A4A4A;color:#fff;border:0;border-radius:10px;padding:13px 26px;font-size:15px}" +
                  "</style></head><body><div class=\"b\"><h1>" + htmlEsc(info.first) + "</h1><p>" + htmlEsc(info.second) + "</p>" +
                  "<p class=\"u\">" + htmlEsc(url) + "</p>" +
                  "<button onclick=\"r()\">Try again</button></div>" +
                  "<script>function r(){location.href=" + js + "}</script></body></html>"
              return "data:text/html;charset=utf-8," + java.net.URLEncoder.encode(html, "UTF-8").replace("+", "%20")
          }

          fun shownUrl(t: BrowserTab?): String {
              val u = t?.url ?: ""
              return if (u == "about:blank") "" else u
          }

          fun normUrl(u: String): String = u.substringBefore('#').trimEnd('/').lowercase()

          object Favicons {
              private val map = mutableStateMapOf<String, ImageBitmap>()
              private val requested = HashSet<String>()
              private val failed = HashMap<String, Long>()
              private val main = Handler(Looper.getMainLooper())

              fun get(key: String): ImageBitmap? = map[key]

              fun request(key: String) {
                  if (key.isBlank() || key == "newtab" || key == "other") return
                  synchronized(requested) {
                      val ft = failed[key]
                      if (ft != null && System.currentTimeMillis() - ft < 600000L) return
                      if (!requested.add(key)) return
                  }
                  Thread {
                      var bmp = readDisk(key)
                      if (bmp == null) {
                          bmp = download("https://" + key + "/favicon.ico")
                          if (bmp == null || bmp.width < 16) {
                              bmp = download("https://www.google.com/s2/favicons?sz=64&domain=" + key)
                          }
                          if (bmp != null) writeDisk(key, bmp)
                      }
                      val b = bmp
                      if (b != null) {
                          val ib = b.asImageBitmap()
                          main.post { map[key] = ib }
                      } else {
                          synchronized(requested) {
                              requested.remove(key)
                              failed[key] = System.currentTimeMillis()
                          }
                      }
                  }.start()
              }

              private fun favDir(): java.io.File {
                  val d = java.io.File(Gecko.appCtx.filesDir, "favicons")
                  d.mkdirs()
                  return d
              }

              private fun favFile(key: String): java.io.File =
                  java.io.File(favDir(), key.replace(Regex("[^A-Za-z0-9.-]"), "_") + ".png")

              private fun readDisk(key: String): Bitmap? {
                  return try {
                      val f = favFile(key)
                      if (f.exists()) BitmapFactory.decodeFile(f.path) else null
                  } catch (e: Exception) {
                      null
                  }
              }

              private fun writeDisk(key: String, b: Bitmap) {
                  try {
                      java.io.FileOutputStream(favFile(key)).use { out ->
                          b.compress(Bitmap.CompressFormat.PNG, 100, out)
                      }
                      val files = favDir().listFiles() ?: return
                      if (files.size > 100) {
                          val sorted = files.sortedBy { it.lastModified() }
                          for (i in 0 until sorted.size - 100) sorted[i].delete()
                      }
                  } catch (e: Exception) {
                  }
              }

              private fun download(u: String): Bitmap? {
                  return try {
                      val c = java.net.URL(u).openConnection() as java.net.HttpURLConnection
                      c.connectTimeout = 8000
                      c.readTimeout = 8000
                      c.instanceFollowRedirects = true
                      c.setRequestProperty("User-Agent", DEFAULT_UA)
                      try {
                          if (c.responseCode in 200..299) c.inputStream.use { BitmapFactory.decodeStream(it) } else null
                      } finally {
                          c.disconnect()
                      }
                  } catch (e: Exception) {
                      null
                  }
              }
          }

          class TabManager(private val runtime: GeckoRuntime) {
              val tabs = mutableStateListOf<BrowserTab>()
              var current by mutableStateOf<BrowserTab?>(null)
              var maxBg by mutableStateOf(4)
              var ctxMenu by mutableStateOf<CtxInfo?>(null)
              var choice by mutableStateOf<ChoiceReq?>(null)
              var js by mutableStateOf<JsReq?>(null)
              var fullscreen by mutableStateOf(false)
              var fullscreenTab = 0
              var home by mutableStateOf(true)
              val closingIds = mutableStateListOf<Int>()
              private val closeJobs = HashMap<Int, Runnable>()
              var view: GeckoView? = null
              private var counter = 0
              private var closing = false
              private val main = Handler(Looper.getMainLooper())
              private val saveRunnable = Runnable { saveNow() }
              private val idleTick = object : Runnable {
                  override fun run() {
                      discardIdle()
                      main.postDelayed(this, 60000L)
                  }
              }

              init {

                  main.postDelayed(idleTick, 60000L)
              }

              private fun thumbFile(id: Int): java.io.File {
                  val dir = java.io.File(Gecko.appCtx.filesDir, "thumbs")
                  dir.mkdirs()
                  return java.io.File(dir, "t" + id + ".jpg")
              }

              fun moveGroupBy(key: String, delta: Int) {
                  val groups = groupTabs(tabs.filter { !it.preview })
                  val i = groups.indexOfFirst { it.key == key }
                  val j = i + delta
                  if (i < 0 || j < 0 || j >= groups.size) return
                  val order = groups.toMutableList()
                  val g = order.removeAt(i)
                  order.add(j, g)
                  val next = ArrayList<BrowserTab>()
                  for (x in order) next.addAll(x.tabs)
                  next.addAll(tabs.filter { it.preview })
                  if (next.size != tabs.size) return
                  for (k in next.indices) {
                      if (tabs[k] !== next[k]) tabs[k] = next[k]
                  }
                  persist()
              }

              fun persist() {
                  if (closing) return
                  main.removeCallbacks(saveRunnable)
                  main.postDelayed(saveRunnable, 700L)
              }

              fun saveNow() {
                  if (closing) return
                  try {
                      val arr = JSONArray()
                      for (t in tabs) {
                      if (t.preview) continue
                      val o = JSONObject()
                      o.put("id", t.id)
                          o.put("u", t.url)
                          o.put("t", t.title)
                          o.put("o", t.openerId)
                          arr.put(o)
                      }
                      val root = JSONObject()
                      val curTab = current
                      root.put("cur", if (curTab == null) 0 else if (curTab.preview) curTab.openerId else curTab.id)
                      root.put("next", counter)
                      root.put("list", arr)
                      Gecko.appCtx.getSharedPreferences("browser", Context.MODE_PRIVATE)
                          .edit().putString("tabs", root.toString()).apply()
                  } catch (e: Exception) {
                      Log.d("GreyScript", "save tabs failed: " + e)
                  }
              }

              fun restore(): Boolean {
                  try {
                      val raw = Gecko.appCtx.getSharedPreferences("browser", Context.MODE_PRIVATE)
                          .getString("tabs", null) ?: return false
                      val root = JSONObject(raw)
                      val arr = root.getJSONArray("list")
                      if (arr.length() == 0) return false
                      var maxId = root.optInt("next", 0)
                      for (i in 0 until arr.length()) {
                          val o = arr.getJSONObject(i)
                          val id = o.getInt("id")
                          val t = BrowserTab(id, null, o.optInt("o", 0))
                          t.url = o.optString("u")
                          t.title = o.optString("t")
                          val f = thumbFile(id)
                          if (f.exists()) {
                              val bmp = BitmapFactory.decodeFile(f.path)
                              if (bmp != null) t.thumb = bmp.asImageBitmap()
                          }
                          if (id > maxId) maxId = id
                          tabs.add(t)
                      }
                      counter = maxId
                      val curId = root.optInt("cur", 0)
                      val c = tabs.firstOrNull { it.id == curId } ?: tabs.last()
                      current = c
                      c.lastUsed = System.currentTimeMillis()
                      val seenKeys = HashSet<String>()
                      for (x in listOf(c) + tabs.filter { it !== c }) {
                          if (x.url.isBlank() || x.url.startsWith("about:")) continue
                          if (!seenKeys.add(normUrl(x.url))) removeTab(x, false)
                      }
                      return true
                  } catch (e: Exception) {
                      Log.d("GreyScript", "restore failed: " + e)
                      tabs.clear()
                      current = null
                      return false
                  }
              }

              // Same site as the reference tab: right below it. Other site: end of that site's group, or the end of the list.
              private fun insertIndexFor(url: String, ref: BrowserTab? = current): Int {
                  val key = keyOfUrl(url)
                  if (ref != null && !ref.preview) {
                      val ci = tabs.indexOf(ref)
                      if (ci >= 0 && groupKeyOf(ref) == key) return ci + 1
                  }
                  var last = -1
                  for (i in tabs.indices) {
                      val x = tabs[i]
                      if (!x.preview && groupKeyOf(x) == key) last = i
                  }
                  return if (last >= 0) last + 1 else tabs.size
              }

              fun newTab(url: String, makeCurrent: Boolean = true, openerId: Int = 0): BrowserTab {
                  var opener = openerId
                  if (url.isNotBlank() && !url.startsWith("about:")) {
                      val key = normUrl(url)
                      val dups = tabs.filter { !it.preview && it.url.isNotBlank() && normUrl(it.url) == key }
                      val cd = dups.firstOrNull { it === current }
                      if (!makeCurrent && cd != null) return cd
                      for (d in dups) {
                          if (d.id == opener) opener = d.openerId
                          if (d === current) current = null
                          removeTab(d, false)
                      }
                  }
                  if (makeCurrent) home = false
                  counter += 1
                  val t = BrowserTab(counter, null, opener)
                  t.url = url
                  tabs.add(insertIndexFor(url).coerceIn(0, tabs.size), t)
                  ensureLive(t, url)
                  Capture.event("TAB", "new tab" + t.id + " " + url + (if (opener != 0) " opener=tab" + opener else ""))
                  if (makeCurrent || current == null) select(t)
                  enforceLimit()
                  persist()
                  return t
              }

              private fun dedupe(t: BrowserTab) {
              if (t.preview) return
                  if (t.url.isBlank() || t.url.startsWith("about:")) return
                  if (!tabs.contains(t)) return
                  val key = normUrl(t.url)
                  val dups = tabs.filter { it !== t && !it.preview && it.url.isNotBlank() && normUrl(it.url) == key }
                  if (dups.isEmpty()) return
                  if (t !== current && dups.any { it === current }) {
                      removeTab(t, false)
                      return
                  }
                  for (d in dups) {
                      if (t.openerId == d.id) t.openerId = d.openerId
                      removeTab(d, false)
                  }
              }

              private fun ensureLive(t: BrowserTab, initialUrl: String? = null) {
              val old = t.session
              if (old != null) {
                  if (old.isOpen) return
                  t.session = null
              }
                  val s = GeckoSession()
                  t.session = s
                  attach(t, s)
                  s.open(runtime)
                  val st = t.savedState
                  if (initialUrl != null) {
                      s.loadUri(initialUrl)
                  } else if (st != null) {
                      s.restoreState(st)
                  } else if (t.url.isNotBlank() && t.url != "about:blank") {
                      s.loadUri(t.url)
                  }
              }

              fun showJs(r: JsReq) {
                  val old = js
                  if (old != null) {
                      try {
                          old.result.complete(old.cancel())
                      } catch (e: Exception) {
                      }
                  }
                  js = r
              }

              fun exitFullscreen() {
                  val ft = tabs.firstOrNull { it.id == fullscreenTab }
                  try {
                      ft?.session?.exitFullScreen()
                  } catch (e: Exception) {
                  }
                  fullscreen = false
                  fullscreenTab = 0
              }

              fun reloadTab(t: BrowserTab) {
                  val eu = t.errorUrl
                  if (eu != null) t.session?.loadUri(eu) else t.session?.reload()
              }

              fun select(t: BrowserTab) {
                  if (fullscreen && fullscreenTab != t.id) exitFullscreen()
                  val prev = current
                  Log.d("GreyScript", "select tab " + t.id + " " + t.url + " session=" + System.identityHashCode(t.session))
                  if (prev != null && prev !== t) prev.lastUsed = System.currentTimeMillis()
                  ensureLive(t)
                  t.lastUsed = System.currentTimeMillis()
                  current = t
                  Capture.event("TAB", "select tab" + t.id + " " + t.url)
                  if (prev != null && prev !== t && prev.preview) removeTab(prev, false)
                  enforceLimit()
                  persist()
              }

              fun load(t: BrowserTab, url: String) {
              home = false
              ensureLive(t)
                  t.session?.loadUri(url)
              }

              fun openerOf(t: BrowserTab): BrowserTab? =
                  if (t.openerId == 0) null else tabs.firstOrNull { it.id == t.openerId }

              fun closeToOpener(t: BrowserTab) {
                  val op = openerOf(t)
                  if (op != null) select(op)
                  close(t)
              }

              fun close(t: BrowserTab) {
                  removeTab(t, true)
              }

              private fun removeTab(t: BrowserTab, keepOne: Boolean) {
                  val i = tabs.indexOf(t)
                  if (i < 0) return
                  Capture.event("TAB", "close tab" + t.id + " " + t.url)
                  tabs.removeAt(i)
                  val s = t.session
                  t.session = null
                  try {
                      thumbFile(t.id).delete()
                  } catch (e: Exception) {
                  }
                  if (tabs.isEmpty()) {
                      current = null
                      if (keepOne) home = true
                  } else if (current === t) {
                      select(tabs[minOf(i, tabs.size - 1)])
                  }
                  if (s != null) main.postDelayed({ s.close() }, 400)
                  persist()
              }

              fun closeAll() {
                  closing = true
                  main.removeCallbacks(saveRunnable)
                  main.removeCallbacks(idleTick)
                  for (t in tabs.toList()) {
                      t.session?.close()
                      t.session = null
                  }
                  tabs.clear()
                  current = null
              }

              fun requestClose(t: BrowserTab) {
                  if (closingIds.contains(t.id)) return
                  closingIds.add(t.id)
                  val r = Runnable {
                      closeJobs.remove(t.id)
                      closingIds.remove(t.id)
                      close(t)
                  }
                  closeJobs[t.id] = r
                  main.postDelayed(r, 2000L)
              }

              fun undoClose(t: BrowserTab) {
                  val r = closeJobs.remove(t.id)
                  if (r != null) main.removeCallbacks(r)
                  closingIds.remove(t.id)
              }

              private fun recover(t: BrowserTab, dead: GeckoSession) {
                  if (t.session !== dead) return
                  Log.d("GreyScript", "session died for tab " + t.id + " " + t.url)
                  if (fullscreenTab == t.id) {
                      fullscreen = false
                      fullscreenTab = 0
                  }
                  t.session = null
                  t.loading = false
                  t.progress = 0
                  t.extId = -1
                  val now = System.currentTimeMillis()
                  if (now - t.lastRecover < 5000L) t.savedState = null
                  t.lastRecover = now
                  if (t === current && !home && t.url.isNotBlank() && t.url != "about:blank") ensureLive(t)
              }

              fun refreshView() {
                  val c = current
                  if (c == null || c.session == null) return
                  main.postDelayed({
                      ensureLive(c)
                      val v = view
                      val s = c.session
                      if (v != null && s != null) {
                          v.releaseSession()
                          v.setSession(s)
                      }
                  }, 200L)
              }

              fun openPreview(url: String, openerId: Int): BrowserTab {
                  var op = openerId
                  val c = current
                  if (c != null && c.preview) op = c.openerId
                  counter += 1
                  val t = BrowserTab(counter, null, op)
                  t.preview = true
                  t.url = url
                  tabs.add(t)
                  ensureLive(t, url)
                  home = false
                  select(t)
                  return t
              }

              fun keep(t: BrowserTab) {
                  t.preview = false
                  dedupe(t)
                  val ref = openerOf(t) ?: current
                  val i = tabs.indexOf(t)
                  if (i >= 0) {
                      tabs.removeAt(i)
                      tabs.add(insertIndexFor(t.url, if (ref === t) null else ref).coerceIn(0, tabs.size), t)
                  }
                  persist()
              }

              fun closePreviewTab(t: BrowserTab) {
                  if (openerOf(t) != null) closeToOpener(t) else close(t)
              }

              fun cycleMaxBg() {
                  val opts = listOf(2, 4, 6, 8, 10)
                  val i = opts.indexOf(maxBg)
                  maxBg = opts[(i + 1) % opts.size]
                  Gecko.appCtx.getSharedPreferences("browser", Context.MODE_PRIVATE)
                      .edit().putInt("maxbg", maxBg).apply()
                  enforceLimit()
              }

              private fun discard(t: BrowserTab) {
                  if (t === current) return
                  val s = t.session ?: return
                  t.session = null
                  t.extId = -1
                  t.loading = false
                  t.progress = 0
                  main.postDelayed({ s.close() }, 150)
              }

              private fun enforceLimit() {
                  val cur = current
                  val bg = tabs.filter { it !== cur && it.session != null }.sortedBy { it.lastUsed }
                  var extra = bg.size - maxBg
                  for (b in bg) {
                      if (extra <= 0) break
                      if (!b.loading) {
                          discard(b)
                          extra--
                      }
                  }
              }

              private fun discardIdle() {
                  val now = System.currentTimeMillis()
                  for (t in tabs.toList()) {
                      if (t !== current && t.session != null && !t.loading && now - t.lastUsed > 1200000L) {
                          discard(t)
                      }
                  }
              }

              fun trim(level: Int) {
                  val keep = if (level >= 40) 1 else if (level >= 15) 2 else return
                  val cur = current
                  val bg = tabs.filter { it !== cur && it.session != null }.sortedByDescending { it.lastUsed }
                  for (i in bg.indices) {
                      if (i >= keep) discard(bg[i])
                  }
              }

              fun capture(t: BrowserTab) {
                  val v = view ?: return
                  if (home || t.session == null) return
                  if (current !== t) return
                  if (v.session !== t.session) return
                  try {
                      v.capturePixels().accept(
                          { bmp ->
                              try {
                                  if (bmp != null && bmp.width > 0 && bmp.height > 0) {
                                      Thread {
                                          try {
                                              val side = minOf(bmp.width, bmp.height)
                                              val sq = Bitmap.createBitmap(bmp, 0, 0, side, side)
                                              val sc = Bitmap.createScaledBitmap(sq, 240, 240, true)
                                              val ib = sc.asImageBitmap()
                                              main.post { t.thumb = ib }
                                              val f = thumbFile(t.id)
                                              java.io.FileOutputStream(f).use { out ->
                                                  sc.compress(Bitmap.CompressFormat.JPEG, 75, out)
                                              }
                                              Backup.schedule()
                                          } catch (e: Exception) {
                                              Log.d("GreyScript", "thumb failed: " + e)
                                          }
                                      }.start()
                                  }
                              } catch (e: Exception) {
                                  Log.d("GreyScript", "thumb failed: " + e)
                              }
                          },
                          { e -> Log.d("GreyScript", "capture failed: " + e) }
                      )
                  } catch (e: Exception) {
                      Log.d("GreyScript", "capture threw: " + e)
                  }
              }

              private fun attach(t: BrowserTab, s: GeckoSession) {
                  s.navigationDelegate = object : GeckoSession.NavigationDelegate {
                      override fun onLocationChange(
                          session: GeckoSession,
                          url: String?,
                          perms: MutableList<GeckoSession.PermissionDelegate.ContentPermission>,
                          hasUserGesture: Boolean
                      ) {
                          val nu = url ?: ""
                          val eu = t.errorUrl
                          if (eu != null && isErrorUri(nu)) {
                              t.url = eu
                          } else {
                              t.errorUrl = null
                              t.url = nu
                          }
                          Capture.event("NAV", "tab" + t.id + " location " + nu + (if (hasUserGesture) " (user gesture)" else ""))
                          dedupe(t)
                          Favicons.request(groupKeyOf(t))
                          persist()
                      }

                      override fun onLoadError(
                          session: GeckoSession,
                          uri: String?,
                          error: WebRequestError
                      ): GeckoResult<String>? {
                          val failed = uri ?: ""
                          if (failed.isBlank() || isErrorUri(failed)) return null
                          Capture.event("NAV", "tab" + t.id + " LOAD ERROR " + failed + " code=" + error.code)
                          t.errorUrl = failed
                          return GeckoResult.fromValue(errorPageUri(failed, error.code))
                      }

                      override fun onCanGoBack(session: GeckoSession, canGoBack: Boolean) {
                          t.canGoBack = canGoBack
                      }

                      override fun onLoadRequest(
                          session: GeckoSession,
                          request: GeckoSession.NavigationDelegate.LoadRequest
                      ): GeckoResult<AllowOrDeny>? {
                          Capture.event("NAV", "tab" + t.id + " load request " + request.uri + (if (request.target == GeckoSession.NavigationDelegate.TARGET_WINDOW_NEW) " [new window]" else ""))
                          if (request.target == GeckoSession.NavigationDelegate.TARGET_WINDOW_NEW) {
                              newTab(request.uri, true, t.id)
                              return GeckoResult.fromValue(AllowOrDeny.DENY)
                          }
                          return null
                      }
                  }
                  s.progressDelegate = object : GeckoSession.ProgressDelegate {
                      override fun onPageStart(session: GeckoSession, url: String) {
                          Capture.event("NAV", "tab" + t.id + " page start " + url)
                          t.loading = true
                          t.progress = 8
                          t.title = ""
                          val eu0 = t.errorUrl
                          if (eu0 != null && isErrorUri(url)) t.url = eu0 else t.url = url
                          t.thumb = null
                          t.scrollY = 0
                          if (HideTool.active && HideTool.tabId == t.id) HideTool.deactivateLocal()
                          t.seen.clear()
                      }

                      override fun onProgressChange(session: GeckoSession, progress: Int) {
                          t.progress = maxOf(progress, 8)
                          t.loading = progress < 100
                      }

                      override fun onPageStop(session: GeckoSession, success: Boolean) {
                          Capture.event("NAV", "tab" + t.id + " page stop " + (if (success) "ok" else "FAILED") + " " + t.url + " title=\"" + t.title + "\"")
                          t.loading = false
                          t.progress = 0
                          if (success && t.errorUrl == null && !t.url.startsWith("moz-extension:")) HistoryStore.add(t.url, t.title)
                          main.postDelayed({ capture(t) }, 800)
                          persist()
                      }

                      override fun onSessionStateChange(
                          session: GeckoSession,
                          sessionState: GeckoSession.SessionState
                      ) {
                          t.savedState = sessionState
                      }
                  }
                  s.promptDelegate = object : GeckoSession.PromptDelegate {
                      override fun onChoicePrompt(
                          session: GeckoSession,
                          prompt: GeckoSession.PromptDelegate.ChoicePrompt
                      ): GeckoResult<GeckoSession.PromptDelegate.PromptResponse>? {
                          val res = GeckoResult<GeckoSession.PromptDelegate.PromptResponse>()
                          val old = choice
                          if (old != null) {
                              try {
                                  old.result.complete(old.prompt.dismiss())
                              } catch (e: Exception) {
                              }
                          }
                          choice = ChoiceReq(prompt, res)
                          return res
                      }

                      override fun onAlertPrompt(
                          session: GeckoSession,
                          prompt: GeckoSession.PromptDelegate.AlertPrompt
                      ): GeckoResult<GeckoSession.PromptDelegate.PromptResponse>? {
                          val res = GeckoResult<GeckoSession.PromptDelegate.PromptResponse>()
                          Capture.event("PROMPT", "tab" + t.id + " alert: " + (prompt.message ?: ""))
                          showJs(JsReq(0, prompt.message ?: "", "", res, { prompt.dismiss() }, { prompt.dismiss() }))
                          return res
                      }

                      override fun onButtonPrompt(
                          session: GeckoSession,
                          prompt: GeckoSession.PromptDelegate.ButtonPrompt
                      ): GeckoResult<GeckoSession.PromptDelegate.PromptResponse>? {
                          val res = GeckoResult<GeckoSession.PromptDelegate.PromptResponse>()
                          Capture.event("PROMPT", "tab" + t.id + " confirm: " + (prompt.message ?: ""))
                          showJs(
                              JsReq(
                                  1, prompt.message ?: "", "", res,
                                  { prompt.confirm(GeckoSession.PromptDelegate.ButtonPrompt.Type.POSITIVE) },
                                  { prompt.confirm(GeckoSession.PromptDelegate.ButtonPrompt.Type.NEGATIVE) }
                              )
                          )
                          return res
                      }

                  if (old != null) {
                      try {
                          old.close()
                      } catch (e: Exception) {
                      }
                  }
                  val s = GeckoSession()
                  popupExt = ext
                  popupTitle = ext.metaData?.name ?: "Extension"
                  popupSession = s
                  return GeckoResult.fromValue(s)
              }

              fun closePopup() {
                  val s = popupSession ?: return
                  popupSession = null
                  popupExt = null
                  main.postDelayed({
                      try {
                          s.close()
                      } catch (e: Exception) {
                      }
                  }, 300L)
              }

              fun activate(s: GeckoSession?) {
                  val c = rt?.webExtensionController ?: return
                  if (s == null || s === activeSession) return
                  try {
                      val old = activeSession
                      if (old != null && old.isOpen) c.setTabActive(old, false)
                      c.setTabActive(s, true)
                      activeSession = s
                  } catch (e: Exception) {
                      Log.e("GreyExt", "setTabActive failed", e)
                  }
              }
          }

          @Composable
          fun ExtPopupOverlay() {
              val s = Exts.popupSession ?: return

              LaunchedEffect(s) {

                      try {
                          val r = Exts.rt
                          if (r != null) s.open(r)
                      } catch (e: Exception) {
                          Log.e("GreyExt", "popup open failed", e)

                  Log.d("GreyExt", "popup session open=" + s.isOpen)
                  ready = true

                  Modifier
                      .fillMaxSize()
                      .background(BG)
                      .pointerInput(Unit) { detectTapGestures { } }
              ) {

                      TextButton(onClick = { Exts.closePopup() }) { Text("←", color = Color.White, fontSize = 20.sp) }

                          TextButton(onClick = { Exts.openOptions(pe) }) { Text("Dashboard", color = Color.White) }
                      }
                  }
                  Box(Modifier.weight(1f).fillMaxWidth()) {
                      AndroidView(
                          factory = { c -> GeckoView(c) },
                          update = { v ->
                              if (ready && s.isOpen && v.session !== s) {
                                  try {
                                      v.setSession(s)
                                  } catch (e: Exception) {
                                      Log.e("GreyExt", "popup setSession failed", e)
                                  }
                              }
                          },
                          onRelease = { v ->
                              try {
                                  v.releaseSession()
                              } catch (e: Exception) {
                              }
                          },
                          modifier = Modifier.fillMaxSize()
                      )
                  }
              }
          }

          class MainActivity : ComponentActivity() {
              private lateinit var tabs: TabManager
              private var started = false

              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  enableEdgeToEdge()
                  Gecko.appCtx = applicationContext
                  if (Backup.hasAccess()) {
                      startApp(true)
                  } else {
                      setContent {
                          AccessGate(
                              onAllow = { Backup.openSettings(this) },
                              onSkip = { startApp(false) }
                          )
                      }
                  }
              }

              override fun onResume() {
                  super.onResume()
                  if (!started && Backup.hasAccess()) startApp(true)
              }

              private fun startApp(useBackup: Boolean) {
                  if (started) return
                  started = true
                  if (useBackup) Backup.start()
                  val runtime = Gecko.runtime(this)
                  Thread { Updater.autoCheck() }.start()
                  tabs = TabManager(runtime)
                  TabBind.tabs = tabs
                  Exts.tabs = tabs
                  tabs.restore()
                  setContent { App(tabs) }
              }

              override fun onStart() {
                  super.onStart()
                  if (::tabs.isInitialized) tabs.refreshView()
              }

              override fun onStop() {
                  if (::tabs.isInitialized) {
                      tabs.saveNow()
                      Backup.flush()
                  }
                  super.onStop()
              }

              override fun onTrimMemory(level: Int) {
                  super.onTrimMemory(level)
                  if (::tabs.isInitialized) tabs.trim(level)
              }

              override fun onDestroy() {
                  TabBind.tabs = null
                  Exts.tabs = null
                  if (::tabs.isInitialized) {
                      tabs.saveNow()
                      tabs.closeAll()
                      Backup.finish()
                  }
                  super.onDestroy()
              }
          }

          fun Context.findActivity(): android.app.Activity? {
              var c: Context? = this
              while (c is android.content.ContextWrapper) {
                  if (c is android.app.Activity) return c
                  c = c.baseContext
              }
              return null
          }

          fun normalize(s: String): String {
              val t = s.trim()
              return when {
                  t.startsWith("http://") || t.startsWith("https://") ||
                      t.startsWith("about:") || t.startsWith("file:") -> t
                  t.contains(" ") || !t.contains(".") ->
                      "https://www.google.com/search?q=" + java.net.URLEncoder.encode(t, "UTF-8")
                  else -> "https://" + t
              }
          }

                          }
                      }
                  }
                  BackHandler(screen != "browser") {
                      screen = when (screen) {
                          "guide", "editor" -> "scripts"
                          else -> "browser"
                      }
                  }
                  LaunchedEffect(Exts.navTick) {
                      if (Exts.navTick > 0) screen = "browser"
                  }
                  Column(
                      Modifier
                          .fillMaxSize()
                          .background(BG)

                              onElements = { screen = "elements" },

                          if (screen != "browser") {

                              ) {

                                          }

                                          tabs,

          class ChoiceRow(val c: GeckoSession.PromptDelegate.ChoicePrompt.Choice, val header: Boolean, val indent: Boolean)

          fun flattenChoices(arr: Array<GeckoSession.PromptDelegate.ChoicePrompt.Choice>?): List<ChoiceRow> {
              val out = ArrayList<ChoiceRow>()
              if (arr == null) return out
              for (c in arr) {
                  val kids = c.items
                  if (kids != null && kids.isNotEmpty()) {
                      out.add(ChoiceRow(c, true, false))
                      for (k in kids) out.add(ChoiceRow(k, false, true))
                  } else {
                      out.add(ChoiceRow(c, false, false))
                  }
              }
              return out
          }

          @Composable
          fun ChoiceDialog(tabs: TabManager) {
              val req = tabs.choice ?: return
              val p = req.prompt
              val rows = remember(req) { flattenChoices(p.choices) }
              val multi = p.type == GeckoSession.PromptDelegate.ChoicePrompt.Type.MULTIPLE
              val picked = remember(req) {
                  val l = mutableStateListOf<String>()
                  for (r in rows) {
                      if (!r.header && r.c.selected) l.add(r.c.id)
                  }
                  l
              }
              fun finish(resp: () -> GeckoSession.PromptDelegate.PromptResponse) {
                  tabs.choice = null
                  try {
                      req.result.complete(resp())
                  } catch (e: Exception) {
                      Log.d("GreyScript", "choice complete failed: " + e)
                  }
              }
              val heading = (p.title ?: "").ifBlank { (p.message ?: "").ifBlank { "Select" } }
              AlertDialog(
                  onDismissRequest = { finish { p.dismiss() } },
                  containerColor = Bar,
                  title = { Text(heading, color = Soft, fontSize = 14.sp, maxLines = 2, overflow = TextOverflow.Ellipsis) },
                  text = {
                      Column(Modifier.heightIn(max = 380.dp).verticalScroll(rememberScrollState())) {
                          for (r in rows) {
                              if (r.header) {
                                  Text(
                                      r.c.label,
                                      color = Soft,
                                      fontSize = 12.sp,
                                      fontWeight = FontWeight.SemiBold,
                                      modifier = Modifier.padding(top = 10.dp, bottom = 4.dp)
                                  )
                              } else if (r.c.separator) {
                                  Box(Modifier.fillMaxWidth().height(0.5.dp).background(Color(0xFF555555)))
                              } else {
                                  val id = r.c.id
                                  val on = if (multi) picked.contains(id) else r.c.selected
                                  val dis = r.c.disabled
                                  Row(
                                      Modifier
                                          .fillMaxWidth()
                                          .clip(RoundedCornerShape(8.dp))
                                          .background(if (on) Field else Color.Transparent)
                                          .clickable(enabled = !dis) {
                                              if (multi) {
                                                  if (picked.contains(id)) picked.remove(id) else picked.add(id)
                                              } else {
                                                  finish { p.confirm(id) }
                                              }
                                          }
                                          .padding(start = if (r.indent) 18.dp else 8.dp, end = 8.dp, top = 12.dp, bottom = 12.dp),
                                      verticalAlignment = Alignment.CenterVertically
                                  ) {
                                      if (multi) {
                                          Text(if (on) "☑" else "☐", color = if (dis) Color(0xFF777777) else Color.White, fontSize = 16.sp)
                                          Spacer(Modifier.width(10.dp))
                                      }
                                      Text(
                                          r.c.label.ifBlank { " " },
                                          color = if (dis) Color(0xFF777777) else Color.White,
                                          fontSize = 15.sp,
                                          modifier = Modifier.weight(1f)
                                      )
                                      if (!multi && on) Text("✓", color = TmGreen, fontSize = 15.sp)
                                  }
                              }
                          }
                      }
                  },
                  confirmButton = {
                      if (multi) {
                          TextButton(onClick = { finish { p.confirm(picked.toTypedArray()) } }) { Text("OK", color = Color.White) }
                      }
                  },
                  dismissButton = {
                      TextButton(onClick = { finish { p.dismiss() } }) { Text("Cancel", color = Color.White) }
                  }
              )
          }

                      }
                  }

          // Shows the gecko loading screen, builds every main screen once underneath it (so the first real
          // open is already warm), then reveals the app.
          @Composable

              var stage by remember { mutableStateOf(0) }
              LaunchedEffect(Unit) {
                  withFrameNanos { }
                  stage = 1
                  repeat(3) { withFrameNanos { } }
                  stage = 2

                  Box(
                      Modifier
                          .fillMaxSize()
                          .background(BG)
                          .pointerInput(Unit) { detectTapGestures { } }
                  ) {
                      if (stage == 1) {
                          Box(Modifier.fillMaxSize()) { TabManagerScreen(tabs, { }, { }, { _ -> }) }
                          Box(Modifier.fillMaxSize()) { HistoryScreen({ }, { _ -> }) }
                          Box(Modifier.fillMaxSize()) { BookmarksScreen(tabs, { }, { _ -> }) }

                          Box(Modifier.fillMaxSize()) { ElementRulesScreen({ }) }
                          Box(Modifier.fillMaxSize()) { GuideScreen({ }) }
                      }
                      Box(
                          Modifier.fillMaxSize().background(BG),
                          contentAlignment = Alignment.Center
                      ) {
                          Image(
                              painter = painterResource(R.drawable.ic_launcher_foreground),
                              contentDescription = null,
                              modifier = Modifier.size(140.dp)
                          )
                      }
                  }
              }
          }

          // Shows a cheap background-only skeleton for the first frame, then builds the real screen and fades in.
          @Composable
          fun ScreenDeferred(content: @Composable () -> Unit) {
              var started by remember { mutableStateOf(false) }
              var ready by remember { mutableStateOf(false) }
              LaunchedEffect(Unit) {
                  started = true
                  withFrameNanos { }

              onScripts: () -> Unit,

              onLogs: () -> Unit,

              val ctx = LocalContext.current

              var markDlg by remember { mutableStateOf(false) }


              LaunchedEffect(cur?.id, tabs.home) {
                  if (HideTool.active && (tabs.home || cur?.id != HideTool.tabId)) HideTool.stop()
              }
              val bmRev = BookmarkStore.rev
              val curUrl = cur?.url ?: ""
              val marked = remember(bmRev, curUrl) {
                  curUrl.isNotBlank() && curUrl != "about:blank" &&
                      BookmarkStore.load().any { normUrl(it.url) == normUrl(curUrl) }
              }

              LaunchedEffect(active) {
                  if (!active) {
                      editing = false
                      focusManager.clearFocus(true)
                      if (HideTool.active) HideTool.stop()
                      val imm = ctx.getSystemService(Context.INPUT_METHOD_SERVICE) as android.view.inputmethod.InputMethodManager
                      imm.hideSoftInputFromWindow(hostView.windowToken, 0)
                  }
              }

              LaunchedEffect(startEdit) {
                  if (startEdit) {
                      text = TextFieldValue("")
                      editing = true
                      onEditStarted()
                  }
              }

              BackHandler(active) {
                  val c = tabs.current
                  if (tabs.fullscreen) {
                      tabs.exitFullscreen()
                  } else if (editing) {
                      editing = false
                      focusManager.clearFocus()
                  } else if (HideTool.active) {
                      HideTool.stop()
                  } else if (tabs.home) {
                  ctx.findActivity()?.moveTaskToBack(true)
                  } else if (c != null && c.canGoBack) {
                      c.session?.goBack()

                  tabs.closeToOpener(c)
                  } else if (c != null && c.preview) {
                  tabs.close(c)
                  } else {
                  tabs.home = true
                  }
              }

                              Button(onClick = { submit(text.text) }) { Text("Go") }

                              Spacer(Modifier.width(6.dp))

          @OptIn(ExperimentalFoundationApi::class)
          @Composable
          fun HideToolPanel() {
              val density = LocalDensity.current
              var ox by remember { mutableStateOf(Float.NaN) }
              var oy by remember { mutableStateOf(Float.NaN) }
              BoxWithConstraints(Modifier.fillMaxSize()) {
                  val maxW = constraints.maxWidth.toFloat()
                  val maxH = constraints.maxHeight.toFloat()
                  val pw = with(density) { 250.dp.toPx() }
                  val ph = with(density) { 140.dp.toPx() }
                  val gap = with(density) { 80.dp.toPx() }
                  LaunchedEffect(maxW, maxH) {
                      if (ox.isNaN()) {
                          ox = (maxW - pw) / 2f
                          oy = maxH - ph - gap
                      }
                  }
                  val x = if (ox.isNaN()) 0f else ox.coerceIn(0f, maxOf(0f, maxW - pw))
                  val y = if (oy.isNaN()) 0f else oy.coerceIn(0f, maxOf(0f, maxH - ph))
                  val shape = RoundedCornerShape(10.dp)
                  Column(
                      Modifier
                          .offset { IntOffset(x.toInt(), y.toInt()) }
                          .width(250.dp)
                          .clip(shape)
                          .background(Bar)
                          .border(0.5.dp, Color(0xFF666666), shape)
                          .pointerInput(Unit) { detectTapGestures { } }
                  ) {
                      Box(
                          Modifier
                              .fillMaxWidth()
                              .height(20.dp)
                              .background(Color.White)
                              .clipToBounds()
                              .pointerInput(Unit) {
                                  detectDragGestures { change, drag ->
                                      change.consume()
                                      val bx = if (ox.isNaN()) x else ox
                                      val by = if (oy.isNaN()) y else oy
                                      ox = (bx + drag.x).coerceIn(0f, maxOf(0f, maxW - pw))
                                      oy = (by + drag.y).coerceIn(0f, maxOf(0f, maxH - ph))
                                  }
                              },
                          contentAlignment = Alignment.CenterStart
                      ) {
                          Text(
                              "Hide element · drag this bar to move",
                              color = Color.Black,
                              fontSize = 10.sp,
                              lineHeight = 12.sp,
                              fontWeight = FontWeight.Bold,
                              maxLines = 1,
                              modifier = Modifier.padding(horizontal = 8.dp)
                          )
                      }
                      Column(Modifier.padding(8.dp)) {
                          Text(
                              if (HideTool.has) HideTool.desc else HideTool.note,
                              color = Color.White,
                              fontSize = 12.sp,
                              maxLines = 2,
                              overflow = TextOverflow.Ellipsis,
                              modifier = Modifier.heightIn(min = 30.dp)
                          )
                          Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                              ToolBtn("Parent", HideTool.has) { HideTool.cmd("parent") }
                              ToolBtn("Child", HideTool.has) { HideTool.cmd("child") }
                              ToolBtn("Preview", HideTool.has, HideTool.previewing) { HideTool.cmd("preview") }
                          }
                          Spacer(Modifier.height(6.dp))
                          Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                              ToolBtn("Hide", HideTool.has) { HideTool.cmd("hide") }
                              ToolBtn(if (HideTool.pageScope) "For: Page" else "For: Site", true, HideTool.pageScope) {
                                  HideTool.pageScope = !HideTool.pageScope
                              }
                              ToolBtn("Close", true) { HideTool.stop() }
                          }
                      }
                  }
              }
          }

          class TabGroup(val key: String, val label: String, val tabs: List<BrowserTab>)

          fun keyOfUrl(u: String): String {
              if (u.isBlank() || u.startsWith("about:")) return "newtab"
              val h = try { android.net.Uri.parse(u).host ?: "" } catch (e: Exception) { "" }
              val d = h.removePrefix("www.")
              return if (d.isBlank()) "other" else d
          }

          fun groupKeyOf(t: BrowserTab?): String = keyOfUrl(t?.url ?: "")

          fun groupTabs(all: List<BrowserTab>): List<TabGroup> {
              val order = ArrayList<String>()
              val map = HashMap<String, MutableList<BrowserTab>>()
              for (t in all) {
                  val k = groupKeyOf(t)
                  val l = map.getOrPut(k) {
                      order.add(k)
                      mutableListOf()
                  }
                  l.add(t)
              }

              }
          }

          object TabMgrMemory {
              var activeKey: String? = null
              var railScroll = 0
              val listScroll = HashMap<String, Int>()
              var curIdAtOpen: Int? = null
          }

          @Composable
          fun TabManagerScreen(
              tabs: TabManager,
              onBack: () -> Unit,
              onNewTab: () -> Unit,
              onSelect: (BrowserTab) -> Unit
          ) {
              val groups = groupTabs(tabs.tabs.filter { !it.preview })
              val curKey = groupKeyOf(tabs.current)
              val mem = TabMgrMemory
              val startKey = remember {
                  val prev = mem.activeKey
                  val changed = mem.curIdAtOpen != tabs.current?.id
                  mem.curIdAtOpen = tabs.current?.id
                  if (prev != null && !(changed && prev != curKey)) prev else curKey
              }
              var activeKey by remember { mutableStateOf(startKey) }
              var dragKey by remember { mutableStateOf<String?>(null) }
              var dragOffset by remember { mutableStateOf(0f) }
              val heights = remember { mutableStateMapOf<String, Int>() }
              val haptic = LocalHapticFeedback.current
              val active = groups.firstOrNull { it.key == activeKey }
                  ?: groups.firstOrNull { it.key == curKey }
                  ?: groups.firstOrNull()
              val railScroll = rememberScrollState(mem.railScroll)
              val listScroll = remember(active?.key) {
                  androidx.compose.foundation.ScrollState(mem.listScroll[active?.key ?: ""] ?: 0)
              }
              LaunchedEffect(railScroll) {
                  snapshotFlow { railScroll.value }.collect { mem.railScroll = it }
              }
              LaunchedEffect(listScroll) {
                  val k = active?.key ?: ""
                  snapshotFlow { listScroll.value }.collect { mem.listScroll[k] = it }
              }
              LaunchedEffect(active?.key) { mem.activeKey = active?.key }

              Column(Modifier.fillMaxSize().background(BG)) {
                  Header {
                      TextButton(onClick = onBack) { Text("←", color = Color.White, fontSize = 20.sp) }
                      Text("Tab Manager", color = Color.White, fontSize = 18.sp, modifier = Modifier.weight(1f))
                      Box(
                          Modifier
                              .size(38.dp)
                              .clip(CircleShape)
                              .background(Color.White)
                              .clickable { onNewTab() },
                          contentAlignment = Alignment.Center
                      ) {
                          Text("+", color = Color.Black, fontSize = 24.sp)
                      }
                      Spacer(Modifier.width(8.dp))
                  }
                  Body {
                      Box(Modifier.fillMaxWidth().height(0.5.dp).background(Field))
                      BoxWithConstraints(Modifier.weight(1f).fillMaxWidth()) {
                          val railW = (maxWidth * 0.38f).coerceIn(120.dp, 160.dp)
                          Row(Modifier.fillMaxSize()) {
                              Column(
                                  Modifier
                                      .width(railW)
                                      .fillMaxHeight()
                                      .background(Bar)
                                      .verticalScroll(railScroll)
                                      .padding(vertical = 8.dp)
                              ) {
                                  for (g in groups) {
                                      key(g.key) {
                                          val dragging = dragKey == g.key
                                          Box(
                                              Modifier
                                                  .onSizeChanged { heights[g.key] = it.height }
                                                  .zIndex(if (dragging) 1f else 0f)
                                                  .graphicsLayer { translationY = if (dragging) dragOffset else 0f }
                                                  .background(if (dragging) Color(0xFF454545) else Color.Transparent)
                                                  .pointerInput(g.key) {
                                                      detectDragGesturesAfterLongPress(
                                                          onDragStart = {
                                                              dragKey = g.key
                                                              dragOffset = 0f
                                                              haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                                                          },
                                                          onDrag = { change, amount ->
                                                              change.consume()
                                                              dragOffset += amount.y
                                                              val cur = groupTabs(tabs.tabs.filter { !it.preview })
                                                              val i = cur.indexOfFirst { it.key == g.key }
                                                              if (dragOffset > 0f && i >= 0 && i < cur.size - 1) {
                                                                  val nh = (heights[cur[i + 1].key] ?: 160).toFloat()
                                                                  if (dragOffset > nh / 2f) {
                                                                      tabs.moveGroupBy(g.key, 1)
                                                                      dragOffset -= nh
                                                                  }
                                                              } else if (dragOffset < 0f && i > 0) {
                                                                  val ph = (heights[cur[i - 1].key] ?: 160).toFloat()
                                                                  if (-dragOffset > ph / 2f) {
                                                                      tabs.moveGroupBy(g.key, -1)
                                                                      dragOffset += ph
                                                                  }
                                                              }
                                                          },
                                                          onDragEnd = {
                                                              dragKey = null
                                                              dragOffset = 0f
                                                          },
                                                          onDragCancel = {
                                                              dragKey = null
                                                              dragOffset = 0f
                                                          }
                                                      )
                                                  }
                                          ) {
                                              GroupItem(
                                                  g,
                                                  active = g.key == active?.key,
                                                  foreground = g.tabs.any { it === tabs.current },
                                                  onClick = { activeKey = g.key }
                                              )
                                          }
                                      }
                                  }
                              }
                              Box(Modifier.width(0.5.dp).fillMaxHeight().background(Field))
                              Column(
                                  Modifier
                                      .weight(1f)
                                      .fillMaxHeight()
                                      .verticalScroll(listScroll)
                                      .padding(12.dp)
                              ) {
                                  if (active != null) {
                                      Row(
                                          Modifier.padding(start = 4.dp, top = 2.dp, bottom = 10.dp),
                                          verticalAlignment = Alignment.CenterVertically
                                      ) {
                                          FaviconBox(active.key, 18.dp, 4.dp, 10)
                                          Spacer(Modifier.width(8.dp))
                                          Text(
                                              active.label.uppercase(),
                                              color = Soft,
                                              fontSize = 11.sp,
                                              fontWeight = FontWeight.SemiBold,
                                              letterSpacing = 0.5.sp
                                          )
                                      }
                                      for (t in active.tabs) {
                                          key(t.id) {
                                              TabCard(
                                                  t,
                                                  foreground = t === tabs.current,
                                                  closing = tabs.closingIds.contains(t.id),
                                                  onClick = { onSelect(t) },
                                                  onToggleClose = {
                                                      if (tabs.closingIds.contains(t.id)) tabs.undoClose(t) else tabs.requestClose(t)
                                                  }
                                              )
                                          }
                                      }
                                  }
                              }
                          }
                      }
                  }
              }
          }

          class BookGroup(val key: String, val items: List<PageEntry>)

          fun groupBookmarks(all: List<PageEntry>): List<BookGroup> {
              val order = ArrayList<String>()
              val map = HashMap<String, MutableList<PageEntry>>()
              for (e in all) {
                  val k = keyOfUrl(e.url)
                  val l = map.getOrPut(k) {
                      order.add(k)
                      mutableListOf()
                  }
                  l.add(e)
              }
              return order.map { k -> BookGroup(k, map[k] ?: mutableListOf()) }
          }

                          )
                          TextButton(onClick = {
                              if (!busy) {
                                  busy = true
                                  status = "Checking…"
                                  scope.launch {
                                      val n = withContext(Dispatchers.IO) { Updater.checkAll() }
                                      reload()
                                      busy = false
                                      status = if (n == 0) "All scripts are up to date" else n.toString() + " update(s) available"
                                  }
                              }
                          }) { Text("Check updates", color = Color.White, fontSize = 12.sp) }
                          if (updates > 0) {
                              TextButton(onClick = {
                                  if (!busy) {
                                      busy = true
                                      status = "Updating…"
                                      scope.launch {
                                          val n = withContext(Dispatchers.IO) { Updater.applyAll() }
                                          reload()
                                          busy = false
                                          status = n.toString() + " script(s) updated"
                                      }
                                  }
                              }) { Text("Update all", color = Color.White, fontSize = 12.sp) }
                          }
                      }
                      if (status.isNotBlank()) {
                          Text(status, color = Soft, fontSize = 12.sp, modifier = Modifier.padding(horizontal = 12.dp, vertical = 2.dp))
                      }
                      if (scripts.isEmpty()) {
                          Text(

                              modifier = Modifier.padding(12.dp)

                                  onClick = { onEdit(s.id) },
                                  onToggle = { v -> Store.setEnabled(s.id, v); reload() }
                              )
                          }
                      }
                  }
              }
              if (showImport) {
                  ImportDialog(
                      onDismiss = { showImport = false },
                      onImported = { name ->
                          showImport = false
                          reload()
                          status = "Imported " + name
                      }
                  )
              }
          }

          @Composable
          fun ScriptRow(s: Script, onClick: () -> Unit, onToggle: (Boolean) -> Unit) {
              val m = remember(s.code) { parseMeta(s.code) }
              val pat = (m.matches + m.includes).firstOrNull() ?: "no @match, won't run"

                              }
                          }

                  },
                  dismissButton = {
                      TextButton(onClick = { if (!busy) onDismiss() }) { Text("Cancel", color = Color.White) }

          @Composable
          fun EditorScreen(id: String?, onBack: () -> Unit) {
              val scope = rememberCoroutineScope()
              val ctx = LocalContext.current
              val existing = remember { if (id == null) null else Store.load().firstOrNull { it.id == id } }
              var curId by remember { mutableStateOf(existing?.id) }
              var code by remember { mutableStateOf(existing?.code ?: NEW_TEMPLATE) }
              var enabled by remember { mutableStateOf(existing?.enabled ?: true) }
              var srcUrl by remember { mutableStateOf(existing?.sourceUrl ?: "") }
              var remoteVer by remember { mutableStateOf(existing?.remoteVersion ?: "") }
              var savedCode by remember { mutableStateOf(existing?.code ?: NEW_TEMPLATE) }
              var savedEnabled by remember { mutableStateOf(existing?.enabled ?: true) }
              var savedSrc by remember { mutableStateOf(existing?.sourceUrl ?: "") }
              var status by remember { mutableStateOf("") }
              var showDiscard by remember { mutableStateOf(false) }
              var showDelete by remember { mutableStateOf(false) }
              val meta = remember(code) { parseMeta(code) }
              val dirty = code != savedCode || enabled != savedEnabled || srcUrl != savedSrc

              BackHandler(dirty) { showDiscard = true }

              fun save() {
                  val list = Store.load()
                  val sid = curId
                  val target = if (sid == null) null else list.firstOrNull { it.id == sid }
                  if (target == null) {
                      val s = Script(sid ?: Store.newId(), code, enabled, srcUrl, remoteVer)
                      list.add(s)
                      curId = s.id
                  } else {
                      target.code = code
                      target.enabled = enabled
                      target.sourceUrl = srcUrl
                      target.remoteVersion = remoteVer
                  }
                  Store.save(list)
                  savedCode = code
                  savedEnabled = enabled
                  savedSrc = srcUrl
                  status = "Saved. Reload the page to run it."
              }

              Column(Modifier.fillMaxSize()) {
                  Header {
                      TextButton(onClick = { if (dirty) showDiscard = true else onBack() }) {
                          Text("←", color = Color.White, fontSize = 20.sp)
                      }
                      Text(
                          if (curId == null) "New script" else "Edit script",
                          color = Color.White,
                          fontSize = 18.sp,
                          modifier = Modifier.weight(1f)
                      )
                      Button(onClick = {
                          save()
                          toast(ctx, "Saved. Reload the page to run it.")
                          onBack()
                      }) { Text("Save") }
                      Spacer(Modifier.width(6.dp))
                  }
                  Body {
                      Column(
                          Modifier.weight(1f).verticalScroll(rememberScrollState()).padding(12.dp)
                      ) {
                          Text("Parsed from header", color = Color(0xFF888888), fontSize = 11.sp)
                          Spacer(Modifier.height(6.dp))
                          Row(Modifier.horizontalScroll(rememberScrollState())) {
                              InfoChip(meta.name)
                              if (meta.version.isNotBlank()) { Spacer(Modifier.width(6.dp)); InfoChip("v" + meta.version) }
                              Spacer(Modifier.width(6.dp))
                              InfoChip((meta.matches.size + meta.includes.size).toString() + " match")
                              Spacer(Modifier.width(6.dp))
                              InfoChip("run-at " + meta.runAt.removePrefix("document-"))
                              if (meta.excludes.isNotEmpty()) { Spacer(Modifier.width(6.dp)); InfoChip(meta.excludes.size.toString() + " exclude") }
                              if (meta.requires.isNotEmpty()) { Spacer(Modifier.width(6.dp)); InfoChip(meta.requires.size.toString() + " require") }
                          }
                          if (meta.matches.isEmpty() && meta.includes.isEmpty()) {
                              Text(
                                  "No @match or @include: this script will never run.",
                                  color = Danger,
                                  fontSize = 12.sp,
                                  modifier = Modifier.padding(top = 6.dp)
                              )
                          }
                          Spacer(Modifier.height(10.dp))
                          Row(verticalAlignment = Alignment.CenterVertically) {
                              Text("Enabled", color = Color.White, modifier = Modifier.weight(1f))
                              Switch(checked = enabled, onCheckedChange = { enabled = it }, colors = switchColors())
                          }
                          Spacer(Modifier.height(10.dp))
                          Text("Update URL (used if the header has no @updateURL or @downloadURL)", color = Color(0xFF888888), fontSize = 11.sp)
                          Spacer(Modifier.height(6.dp))
                          OutlinedTextField(
                              value = srcUrl,
                              onValueChange = { srcUrl = it },
                              modifier = Modifier.fillMaxWidth(),
                              singleLine = true,
                              textStyle = TextStyle(fontFamily = FontFamily.Monospace, fontSize = 12.sp, color = Color.White),
                              colors = fieldColors()
                          )
                          Spacer(Modifier.height(8.dp))
                          Row(verticalAlignment = Alignment.CenterVertically) {
                              Button(onClick = {
                                  val u = checkUrlFor(meta, srcUrl)
                                  if (u.isBlank()) {
                                      status = "Add an update URL first"
                                  } else {
                                      status = "Checking…"
                                      scope.launch {
                                          val r = withContext(Dispatchers.IO) {
                                              try { httpGet(u) } catch (e: Exception) { "ERR:" + e.message }
                                          }
                                          if (r.startsWith("ERR:")) {
                                              status = r.substring(4)
                                          } else {
                                              val rv = parseMeta(r).version
                                              if (rv.isNotBlank() && compareVersions(rv, meta.version) > 0) {
                                                  remoteVer = rv
                                                  status = "Update " + rv + " available"
                                              } else {
                                                  remoteVer = ""
                                                  status = "Up to date"
                                              }
                                          }
                                      }
                                  }
                              }) { Text("Check") }
                              if (remoteVer.isNotBlank()) {
                                  Spacer(Modifier.width(8.dp))
                                  Button(onClick = {
                                      val u = applyUrlFor(meta, srcUrl)
                                      if (u.isBlank()) {
                                          status = "No update URL"
                                      } else {
                                          status = "Downloading…"
                                          scope.launch {
                                              val r = withContext(Dispatchers.IO) {
                                                  try { httpGet(u) } catch (e: Exception) { "ERR:" + e.message }
                                              }
                                              if (r.startsWith("ERR:")) {
                                                  status = r.substring(4)
                                              } else if (!r.contains("==UserScript==")) {
                                                  status = "Downloaded file has no script header"
                                              } else {
                                                  code = r
                                                  remoteVer = ""
                                                  save()
                                                  status = "Updated to v" + parseMeta(r).version
                                              }
                                          }
                                      }
                                  }) { Text("Update to v" + remoteVer) }
                              }
                          }
                          if (status.isNotBlank()) {
                              Text(status, color = Soft, fontSize = 12.sp, modifier = Modifier.padding(top = 8.dp))
                          }
                          if (curId != null) {
                              Text(
                                  "Delete script",
                                  color = Danger,
                                  fontSize = 14.sp,
                                  modifier = Modifier
                                      .padding(top = 20.dp, bottom = 12.dp)
                                      .clickable { showDelete = true }
                              )
                          }
                          Spacer(Modifier.height(10.dp))
                          OutlinedTextField(
                              value = code,
                              onValueChange = { code = it },
                              modifier = Modifier.fillMaxWidth(),
                              minLines = 12,
                              textStyle = TextStyle(fontFamily = FontFamily.Monospace, fontSize = 12.sp, color = Color.White),
                              colors = fieldColors()
                          )
                      }
                  }
              }
              if (showDiscard) {
                  AlertDialog(
                      onDismissRequest = { showDiscard = false },
                      containerColor = Bar,
                      title = { Text("Discard changes?", color = Color.White) },
                      text = { Text("You have unsaved changes.", color = Soft) },
                      confirmButton = {
                          TextButton(onClick = { showDiscard = false; onBack() }) { Text("Discard", color = Danger) }
                      },
                      dismissButton = {
                          TextButton(onClick = { showDiscard = false }) { Text("Keep editing", color = Color.White) }
                      }
                  )
              }
              if (showDelete) {
                  AlertDialog(
                      onDismissRequest = { showDelete = false },
                      containerColor = Bar,
                      title = { Text("Delete script?", color = Color.White) },
                      text = { Text(meta.name, color = Soft) },
                      confirmButton = {
                          TextButton(onClick = {
                              showDelete = false
                              val sid = curId
                              if (sid != null) Store.remove(sid)
                              onBack()
                          }) { Text("Delete", color = Danger) }
                      },
                      dismissButton = {
                          TextButton(onClick = { showDelete = false }) { Text("Cancel", color = Color.White) }
                      }
                  )
              }
          }

          fun displayName(ctx: Context, uri: android.net.Uri): String {
              var n = ""
              try {
                  ctx.contentResolver.query(uri, null, null, null, null)?.use { c ->
                      if (c.moveToFirst()) {
                          val i = c.getColumnIndex(android.provider.OpenableColumns.DISPLAY_NAME)
                          if (i >= 0) n = c.getString(i) ?: ""
                      }
                  }
              } catch (e: Exception) {
              }
              return n.removeSuffix(".txt").ifBlank { "Imported list" }
          }
